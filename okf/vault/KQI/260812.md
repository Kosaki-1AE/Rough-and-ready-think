Field[0] = (
	ID       = 0;
	Element  = {};
	state    = 0.0;
	Bias     = 0.0;
	Radius   = 0.5;
	Range    = range(-0.5, 0.5);
	Boundary = {-0.5, 0.5};
	Memory   = tau;
	Token    = {};
);

Global[0] = (
	Token          = {};
	Pending        = {};
	Feedback       = {};
	state          = 0.0;
	Target         = 0.0;
	Loss           = 0.0;
	Responsibility = {};
);

distill({}, Memory) = Field.InitialBias;
spread({}, Bias, Memory) = Field.InitialRadius;

＃Local Optimization
import.Port(x, t) = (
	Current.Field = snapshot(Field[t]);

	Current.Field = convert(
		Current.Field.Element               -> History
		tau                                 -> Memory
		distill(History, Memory)            -> Bias
		spread(History, Memory, Bias)       -> Spread
		clamp(Spread, 0.1, 0.5)             -> Radius
		range(Bias - Radius, Bias + Radius) -> Range
		{Range.min, Range.max}              -> Boundary
	);

	Port(x) = convert(
		"$delta_z" -> Input
		"$abln2"   -> Weight
	);

	Next.Field.state =
		Current.Field.state + (
			Current.Field.Bias
			+ Port(x)
			- Current.Field.state
		) / Current.Field.Memory * dt;

	if (Next.Field.state in Current.Field.Range) {
		accept.Port(x);
		status = 0;
		exit = 0;

	} else if (crossed(
		Current.Field.state,
		Next.Field.state,
		Current.Field.Boundary
	)) {
		transition.Port(x);
		status = 0;

	} else {
		reject.Port(x);
		status = 1;
	}

	Pending.Token = tokenize(
		Source -> Current.Field.ID
		Input -> Port(x)
		Before -> Current.Field.state
		After -> Next.Field.state
		Result -> status
		Time -> t
	);
	
	Field[t + 1] = commit(
		Current.Field,
		Next.Field,
		Pending.Element
	);
	
	Global.Pending += Pending.Token;
	
	return status;
);

＃Global Optimization
optimize.Global(t) = (
    Current.Global = snapshot(Global[t]);

    for (Token in Current.Global.Pending) {
        Token.Responsibility = evaluate(
            Token.Change      -> Delta
            Global.Sensitivity -> Weight
            Global.Loss        -> Loss
        );

        Current.Global.Responsibility[Token.Source]
            += Token.Responsibility;
    }

    Next.Global.state = aggregate(
        Current.Global.state,
        Current.Global.Pending
    );

    Next.Global.Loss = loss(
        Next.Global.state,
        Global.Target
    );

    Global[t + 1] = commit(
        Current.Global,
        Next.Global
    );

    return Next.Global.Loss;
);

＃Local -> Global
Token.Responsibility = evaluate(
    Token.Delta       -> Change,
    Global.Sensitivity -> Weight
);

＃Global -> Local
Feedback.Shame = convert(
    Token.Responsibility -> Input,
    Local.Internalization -> Weight,
    Local.Bias            -> Bias
);

Next.Local.state = Current.Local.state + Feedback.Shame / Local.Memory * dt;

if (Feedback.Shame in Local.LearningRange) {
    update.Local;
} else if (crossed(Feedback.Shame, Local.Boundary)) {
    transition.Local;
} else {
    freeze.Local;
}

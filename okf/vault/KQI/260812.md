Field[0] = (
    ID       = 0;
    Element  = {};
    state    = 0.0;
    Bias     = 0.0;
    Radius   = 0.5;
    Range    = range(-0.5, 0.5);
    Boundary = {-0.5, 0.5};
    Memory   = tau;
    Token    = {};
);

Global[0] = (
    Token          = {};
    Pending        = {};
    Feedback       = {};
    state          = 0.0;
    Target         = 0.0;
    Loss           = 0.0;
    Responsibility = {};
);

distill({}, Memory)      = Field.InitialBias;
spread({}, Bias, Memory) = Field.InitialRadius;

＃Local Optimization
import.Port(x, t) = (
    Current.Field = snapshot(Field[t]);

    Current.Field = convert(
        Current.Field.Element               -> History
        tau                                 -> Memory
        distill(History, Memory)             -> Bias
        spread(History, Memory, Bias)        -> Spread
        clamp(Spread, 0.1, 0.5)              -> Radius
        range(Bias - Radius, Bias + Radius)  -> Range
        {Range.min, Range.max}               -> Boundary
    );

    Port(x) = convert(
        "$delta_z" -> Input
        "$abln2"   -> Weight
    );

    Next.Field.state =
        Current.Field.state
        + (
            Current.Field.Bias
            + Port(x)
            - Current.Field.state
        )
        / Current.Field.Memory
        * dt;

    if (Next.Field.state in Current.Field.Range) {
        accept.Port(x);
        result = ACCEPT;
        exit   = 0;

    } else if (crossed(
        Current.Field.state,
        Next.Field.state,
        Current.Field.Boundary
    )) {
        transition.Port(x);
        result = TRANSITION;
        exit   = 0;

    } else {
        reject.Port(x);
        result = REJECT;
        exit   = 1;
    }

    Pending.Token = tokenize(
        Source -> Current.Field.ID
        Input  -> Port(x)
        Before -> Current.Field.state
        After  -> Next.Field.state
        Delta  -> Next.Field.state - Current.Field.state
        Result -> result
        Exit   -> exit
        Time   -> t
    );

    Field[t + 1] = commit(
        Current.Field,
        Next.Field,
        Pending.Token
    );

    Global[t].Pending += Pending.Token;

    return exit;
);

＃Global Optimization
optimize.Global(t) = (
    Current.Global  = snapshot(Global[t]);
    Next.Global     = copy(Current.Global);
    Pending.Feedback = {};

    Next.Global.state = aggregate(
        Current.Global.state,
        Current.Global.Pending
    );

    Next.Global.Loss = loss(
        Next.Global.state,
        Current.Global.Target
    );

    for (Token in Current.Global.Pending) {
        Responsibility = gradient(
            Next.Global.Loss,
            Token
        );

        ResponsibilityMagnitude =
            abs(Responsibility);

        Feedback.Signal = convert(
            Responsibility          -> Input
            Global.ReturnWeight     -> Weight
            Global.FeedbackBias     -> Bias
        );

        Shame = internalize(
            Feedback.Signal                          -> Input
            Field[Token.Source].Internalization      -> Weight
            Field[Token.Source].Bias                 -> Bias
        );

        Pending.Feedback[Token.Source] += (
            Signal                  -> Feedback.Signal
            Shame                   -> Shame
            Responsibility          -> Responsibility
            ResponsibilityMagnitude -> ResponsibilityMagnitude
            Time                    -> t
        );

        Next.Global.Responsibility[Token.Source]
            += Responsibility;
    }

    Next.Global.Feedback = Pending.Feedback;
    Next.Global.Pending  = {};

    Global[t + 1] = commit(
        Current.Global,
        Next.Global
    );

    return Next.Global.Loss;
);

update.Local(id, t + 1) = (
    Current.Local = snapshot(Field[id][t + 1]);
    Feedback      = Global[t + 1].Feedback[id];

    Next.Local.state =
        Current.Local.state
        + Feedback.Shame
        / Current.Local.Memory
        * dt;

    if (Next.Local.state in Current.Local.LearningRange) {
        learn.Local;
        result = LEARN;
        exit   = 0;

    } else if (crossed(
        Current.Local.state,
        Next.Local.state,
        Current.Local.Boundary
    )) {
        transition.Local;
        result = TRANSITION;
        exit   = 0;

    } else {
        freeze.Local;
        result = FREEZE;
        exit   = 1;
    }

    Field[id][t + 2] = commit(
        Current.Local,
        Next.Local,
        result
    );

    return exit;
);

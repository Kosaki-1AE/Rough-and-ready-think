Field[0][0] = (
    ID              = 0;
    Element         = {};
    state           = 0.0;

    Bias            = 0.0;
    Radius          = 0.5;
    Range           = range(-0.5, 0.5);
    Boundary        = {-0.5, 0.5};

    Learning.Radius  = 0.5;
    Learning.Range   = range(-0.5, 0.5);

    Memory          = tau;
    Internalization = 1.0;

    Port            = DEFAULT_PORT;
    Target          = 0.0;

    Token           = {};
    Reframe         = {};
);

Global[0] = (
    Pending        = {};
    Feedback       = {};
    Reframe        = {};

    state          = 0.0;
    Target         = 0.0;
    Loss           = 0.0;

    Return.Weight   = 1.0;
    Feedback.Bias   = 0.0;

    Responsibility = {};
    History        = {};
);

distill({}, Memory) =
    Field.InitialBias;

spread({}, Bias, Memory) =
    Field.InitialRadius;


/* =========================================================
   Local Optimization
   Inputを現在のFieldへ取り込み、判定結果をToken化する
   ========================================================= */

import.Port(id, x, t) = (
    Current.Field = snapshot(
        Field[id][t]
    );

    Current.Field = convert(
        Current.Field.Element                    -> History
        Current.Field.Memory                     -> Memory

        distill(History, Memory)                  -> Bias
        spread(History, Memory, Bias)             -> Spread

        clamp(Spread, 0.1, 0.5)                  -> Radius
        range(Bias - Radius, Bias + Radius)      -> Range
        {Range.min, Range.max}                   -> Boundary
    );

    Port(x) = convert(
        x                      -> Input
        "$delta_z"             -> Difference
        "$abln2"               -> Weight
        Current.Field.Port     -> Method
        Current.Field.Bias     -> Bias
    );

    Next.Field = copy(
        Current.Field
    );

    Next.Field.state =
        Current.Field.state
        + (
            Current.Field.Bias
            + Port(x)
            - Current.Field.state
        )
        / Current.Field.Memory
        * dt;

    if (
        Next.Field.state
        in Current.Field.Range
    ) {
        accept.Port(x);

        result = ACCEPT;
        exit   = 0;

    } else if (
        crossed(
            Current.Field.state,
            Next.Field.state,
            Current.Field.Boundary
        )
    ) {
        transition.Port(x);

        result = TRANSITION;
        exit   = 0;

    } else {
        reject.Port(x);

        result = REJECT;
        exit   = 1;
    }

    Pending.Token = tokenize(
        Source   -> Current.Field.ID
        Input    -> x
        Port     -> Port(x)
        Method   -> Current.Field.Port

        Before   -> Current.Field.state
        After    -> Next.Field.state
        Delta    -> Next.Field.state
                   - Current.Field.state

        Bias     -> Current.Field.Bias
        Range    -> Current.Field.Range
        Boundary -> Current.Field.Boundary
        Target   -> Current.Field.Target

        Result   -> result
        Exit     -> exit
        Time     -> t
    );

    Next.Field.Element += Pending.Token;
    Next.Field.Token    = Pending.Token;

    Field[id][t + 1] = commit(
        Current.Field,
        Next.Field,
        Pending.Token
    );

    Global.Pending = enqueue(
        Global.Pending,
        Pending.Token
    );

    return exit;
);


/* =========================================================
   Global Optimization
   LocalのTokenを集約し、全体Lossと責任を計算する
   ========================================================= */

optimize.Global(t) = (
    Current.Global   = snapshot(
        Global[t]
    );

    Next.Global      = copy(
        Current.Global
    );

    Pending.Feedback = {};
    Pending.Reframe  = {};

    Next.Global.state = aggregate(
        Current.Global.state,
        Current.Global.Pending
    );

    Next.Global.Loss = loss(
        Next.Global.state,
        Current.Global.Target
    );

    for (
        Token in Current.Global.Pending
    ) {
        Responsibility = gradient(
            Next.Global.Loss,
            Token
        );

        ResponsibilityMagnitude =
            abs(Responsibility);

        Feedback.Signal = convert(
            Responsibility               -> Input
            Current.Global.ReturnWeight  -> Weight
            Current.Global.FeedbackBias  -> Bias
        );

        InternalizedFeedback = internalize(
            Feedback.Signal
                -> Input

            Field[Token.Source][t + 1]
                .Internalization
                -> Weight

            Field[Token.Source][t + 1]
                .Bias
                -> Bias
        );

        Pending.Feedback[Token.Source] += (
            Signal
                -> Feedback.Signal

            Internalized
                -> InternalizedFeedback

            Responsibility
                -> Responsibility

            ResponsibilityMagnitude
                -> ResponsibilityMagnitude

            Source
                -> Token.Source

            Token
                -> Token

            Time
                -> t
        );

        Next.Global.Responsibility[
            Token.Source
        ] += Responsibility;

        if (
            Token.Result == REJECT
            || ResponsibilityMagnitude
               > Global.ReframeThreshold
        ) {
            Pending.Reframe[
                Token.Source
            ] += request.Reframe(
                Token
                    -> Evidence

                Feedback.Signal
                    -> Constraint

                Current.Global.Target
                    -> Target

                Token.Boundary
                    -> Boundary

                Token.Time
                    -> Time
            );
        }
    }

    Next.Global.Feedback =
        Pending.Feedback;

    Next.Global.Reframe =
        Pending.Reframe;

    Next.Global.History +=
        Current.Global.Pending;

    Next.Global.Pending = {};

    Global[t + 1] = commit(
        Current.Global,
        Next.Global
    );

    return Next.Global.Loss;
);


/* =========================================================
   Global -> Local
   FeedbackをLocalへ戻して学習可能性を判定する
   ========================================================= */

update.Local(id, t + 1) = (
    Current.Local = snapshot(
        Field[id][t + 1]
    );

    Feedback = Global[t + 1]
        .Feedback[id];

    Reframe.Request = Global[t + 1]
        .Reframe[id];

    Next.Local = copy(
        Current.Local
    );

    Next.Local.state =
        Current.Local.state
        + Feedback.Internalized
        / Current.Local.Memory
        * dt;

    if (
        Next.Local.state
        in Current.Local.LearningRange
    ) {
        learn.Local;

        result = LEARN;
        exit   = 0;

    } else if (
        crossed(
            Current.Local.state,
            Next.Local.state,
            Current.Local.Boundary
        )
    ) {
        transition.Local;

        result = TRANSITION;
        exit   = 0;

    } else {
        freeze.Local;

        result = FREEZE;
        exit   = 1;
    }

    Pending.UpdateToken = tokenize(
        Source     -> id
        Before     -> Current.Local.state
        After      -> Next.Local.state
        Delta      -> Next.Local.state
                      - Current.Local.state

        Feedback   -> Feedback.Signal
        Internalized
                   -> Feedback.Internalized

        Result     -> result
        Exit       -> exit
        Time       -> t + 1
    );

    Field[id][t + 2] = commit(
        Current.Local,
        Next.Local,
        Pending.UpdateToken
    );

    if (
        result == FREEZE
        || Reframe.Request != {}
    ) {
        reframe.Local(
            id,
            t + 2,
            Reframe.Request
        );
    }

    return exit;
);


/* =========================================================
   Reframe
   答え方ではなく、解き方・境界・問い自体を組み替える
   ========================================================= */

reframe.Local(id, t) = (
    Current.Local = snapshot(
        Field[id][t]
    );

    Current.Global = snapshot(
        Global[t - 1]
    );

    Evidence = collect(
        Current.Local.Element,
        Current.Local.Token,
        Current.Global.History,
        Current.Global.Reframe[id]
    );

    Reframe = reformulate(
        Evidence
            -> History

        Current.Local.Target
            -> LocalTarget

        Current.Global.Target
            -> GlobalTarget

        Current.Local.Bias
            -> Bias

        Current.Local.Range
            -> Range

        Current.Local.Boundary
            -> Boundary

        Current.Local.Port
            -> Port

        Current.Local.Memory
            -> Memory
    );

    Next.Local = copy(
        Current.Local
    );

    Next.Local.Bias =
        Reframe.Bias;

    Next.Local.Radius =
        clamp(
            Reframe.Radius,
            0.1,
            0.5
        );

    Next.Local.Range = range(
        Next.Local.Bias
            - Next.Local.Radius,

        Next.Local.Bias
            + Next.Local.Radius
    );

    Next.Local.Boundary = {
        Next.Local.Range.min,
        Next.Local.Range.max
    };

    Next.Local.LearningRadius =
        clamp(
            Reframe.LearningRadius,
            0.1,
            0.5
        );

    Next.Local.LearningRange = range(
        Next.Local.Bias
            - Next.Local.LearningRadius,

        Next.Local.Bias
            + Next.Local.LearningRadius
    );

    Next.Local.Port =
        Reframe.Port;

    Next.Local.Target =
        Reframe.LocalTarget;

    Pending.ReframeToken = tokenize(
        Source         -> id

        PreviousBias   -> Current.Local.Bias
        NextBias       -> Next.Local.Bias

        PreviousRange  -> Current.Local.Range
        NextRange      -> Next.Local.Range

        PreviousPort   -> Current.Local.Port
        NextPort       -> Next.Local.Port

        PreviousTarget -> Current.Local.Target
        NextTarget     -> Next.Local.Target

        Evidence       -> Evidence
        Result         -> REFRAME
        Time           -> t
    );

    Next.Local.Reframe =
        Pending.ReframeToken;

    Next.Local.Element +=
        Pending.ReframeToken;

    Field[id][t + 1] = commit(
        Current.Local,
        Next.Local,
        Pending.ReframeToken
    );

    if (
        Reframe.GlobalTarget
        != Current.Global.Target
    ) {
        propose.GlobalReframe(
            Source   -> id
            Target   -> Reframe.GlobalTarget
            Evidence -> Evidence
            Time     -> t
        );
    }

    return 0;
);


/* =========================================================
   Global Reframe
   十分な根拠がある場合のみ、全体の問いを更新する
   ========================================================= */

reframe.Global(t) = (
    Current.Global = snapshot(
        Global[t]
    );

    Proposals = collect(
        Current.Global.Reframe
    );

    Candidate.Target = distill(
        Proposals,
        Current.Global.History
    );

    TargetChange.Loss = loss(
        Candidate.Target,
        Current.Global.Target
    );

    if (
        validate(
            Candidate.Target,
            Proposals,
            Current.Global.History
        )
    ) {
        Next.Global = copy(
            Current.Global
        );

        Next.Global.Target =
            Candidate.Target;

        result = GLOBAL_REFRAME;
        exit   = 0;

    } else {
        Next.Global = copy(
            Current.Global
        );

        result = KEEP_TARGET;
        exit   = 1;
    }

    Pending.GlobalReframeToken = tokenize(
        Previous.Target
            -> Current.Global.Target

        CandidateTarget
            -> Candidate.Target

        Evidence
            -> Proposals

        Result
            -> result

        Exit
            -> exit

        Time
            -> t
    );

    Next.Global.History +=
        Pending.GlobalReframeToken;

    Global[t + 1] = commit(
        Current.Global,
        Next.Global,
        Pending.GlobalReframeToken
    );

    return exit;
);


/* =========================================================
   Retry
   Reframe後のFieldへ、過去のInputをかけ直す
   ========================================================= */

retry.Port(id, Failed.Token, t) = (
    Current.Field = snapshot(
        Field[id][t]
    );

    Retry.Input = convert(
        Failed.Token.Input
            -> Input

        Failed.Token
            -> History

        Current.Field.Port
            -> NewMethod

        Current.Field.Bias
            -> NewBias

        Current.Field.Target
            -> NewTarget
    );

    Pending.RetryToken = tokenize(
        Source
            -> id

        Parent
            -> Failed.Token

        Input
            -> Retry.Input

        Method
            -> Current.Field.Port

        Target
            -> Current.Field.Target

        Result
            -> RETRY

        Time
            -> t
    );

    Current.Field.Element +=
        Pending.RetryToken;

    return import.Port(
        id,
        Retry.Input,
        t
    );
);

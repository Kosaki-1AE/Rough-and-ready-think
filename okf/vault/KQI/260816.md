#Mine研究/実装 
```c
import.Port(x) = (

	Port(x) = convert(
		Input -> "$delta_z"
		Weight -> "$abln2"
		Memory -> "tau"
	);
	
	Field = (
	    Range    = range(-0.5, 0.5);
	    Boundary = {-0.5, 0.5};
	    Step     = 0.1;
	);
	
	Identity = (
	    Preserve  = 0;
	    Transition = 1;
	    Undefined = -1;
	);

	Field.state += ( Port(x) - Field.state / tau ) * dt;
	
	if (Field.Boundary.min < Port(x) && Port(x) < Field.Boundary.max) {
		Identity = Preserve;
		accept.Port(x);
		continue_action;
		return 0;
		
	} else if (Port(x) == Field.Boundary.min || Port(x) == Field.Boundary.max) {
		Identity = Transition;
		transition.Port(x);
		continue_action;
		return 0;
		
	} else {
		Identity = Undefined;
		reject.Port(x);
		return 1;
	}
}
```

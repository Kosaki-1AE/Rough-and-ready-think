構造木をどう折るかだよなぁ多分w
()が $ab\ln2$ 、{}が $2ab\ln2$ か。レベルを把握するのが $ab\ln2$ なら面白いのよなw つまりは $ab\ln2$ 1つが責任ってことよなw
多分これが成り立つわけで：$R(×)-R(+)=ab\ln2$

根から行くやつが構造木、葉から行くやつが履歴・責任の回収(整合性の一致=$ab\ln2$)ってことなはずで。

```c
typedef int Bool;

＃define FALSE 0
＃define TRUE  1

typedef enum {
    RESULT_ACCEPT = 0,
    RESULT_TRANSITION = 1,
    RESULT_REJECT = 2
} Result;

typedef enum {
    ACTION_PRESERVE = 0,
    ACTION_REFRAME = 1,
    ACTION_REJECT = 2
} Action;

typedef struct {
    double input;      /* delta_z */
    double weight;     /* abln2   */
    double memory;     /* tau      */
} Port;

typedef struct {
    double resolution;

    double state;

    double range_min;
    double range_max;

    double boundary_min;
    double boundary_max;

    double step;
} Field;

typedef struct {
    double before;
    double after;

    Bool identity;
    Result result;

    double range_min;
    double range_max;

    double boundary_min;
    double boundary_max;
} Meaning;

typedef struct {
    double target;
    double loss;
    double threshold;
} Global;

typedef struct {
    Field next_field;
    Meaning meaning;
    Global global;

    Bool identity;
    Result result;
    Action action;
} Evaluation;


/* =========================================================
   Primitive
========================================================= */

static double abs_d(double x)
{
    if (x < 0.0) {
        return -x;
    }

    return x;
}

double ln2(void)
{
    double x = 1.0 / 3.0;
    double x2 = x * x;

    double term = x;
    double sum = 0.0;

    int n;

    for (n = 1; n < 50; n += 2) {
        sum += term / n;
        term *= x2;
    }

    return 2.0 * sum;
}

double responsibility_step(double a, double b)
{
    return a * b * ln2();
}

typedef struct Node {
    int level;
    int branch_count;

    double responsibility;

    struct Node *parent;
    struct Node *left;
    struct Node *right;
} Node;

child.responsibility =
    parent.responsibility
    + a * b * ln2();

/* =========================================================
   Layer 1 : State
   何が変わったか
========================================================= */

static Port convert_port(
    double delta_z,
    double abln2,
    double tau
)
{
    Port port;

    port.input  = delta_z;
    port.weight = abln2;
    port.memory = tau;

    return port;
}


static Field update_field(
    Field current,
    Port port,
    double dt
)
{
    Field next;
    double weighted_input;

    next = current;

    /*
       現段階では
           Port(x) = delta_z * abln2
       として具体化している。

       abln2 の構造木・責任モデルが確定したら、
       この部分を responsibility() 等へ置換可能。
    */
    weighted_input = port.input * port.weight;

    next.state =
        current.state
        + (
            weighted_input
            - current.state
        )
        / port.memory
        * dt;

    return next;
}


/* =========================================================
   Layer 2 : Identity
   変化前後を同じものとして扱えるか
========================================================= */

static Result judge_identity(
    Field field,
    Bool *identity
)
{
    double x;
    double epsilon;

    x = field.state;
    epsilon = 0.000000001;

    if (
        field.boundary_min < x
        &&
        x < field.boundary_max
    ) {
        *identity = TRUE;
        return RESULT_ACCEPT;
    }

    if (
        abs_d(x - field.boundary_min) < epsilon
        ||
        abs_d(x - field.boundary_max) < epsilon
    ) {
        *identity = FALSE;
        return RESULT_TRANSITION;
    }

    *identity = FALSE;
    return RESULT_REJECT;
}


/* =========================================================
   Layer 3 : Meaning
   Identity判定を全体としてどう解釈するか
========================================================= */

static Meaning evaluate_meaning(
    Field current,
    Field next,
    Bool identity,
    Result result
)
{
    Meaning meaning;

    meaning.before = current.state;
    meaning.after  = next.state;

    meaning.identity = identity;
    meaning.result   = result;

    meaning.range_min = current.range_min;
    meaning.range_max = current.range_max;

    meaning.boundary_min = current.boundary_min;
    meaning.boundary_max = current.boundary_max;

    return meaning;
}


static double calculate_loss(
    Meaning meaning,
    double target
)
{
    return abs_d(
        meaning.after
        - target
    );
}


/* =========================================================
   Global
   全体整合を確認する
========================================================= */

static Global evaluate_global(
    Meaning meaning,
    double target,
    double threshold
)
{
    Global global;

    global.target = target;
    global.loss = calculate_loss(
        meaning,
        target
    );
    global.threshold = threshold;

    return global;
}


/* =========================================================
   Action
   全体整合と個別整合から行動を決める
========================================================= */

static Action decide_action(
    Result result,
    Global global
)
{
    if (result == RESULT_REJECT) {
        return ACTION_REJECT;
    }

    if (global.loss <= global.threshold) {
        return ACTION_PRESERVE;
    }

    return ACTION_REFRAME;
}


/* =========================================================
   import.Port(x)
   Core Entry Point
========================================================= */

Evaluation import_port(
    Field current,
    double delta_z,
    double abln2,
    double tau,
    double dt,
    double global_target,
    double global_threshold
)
{
    Evaluation evaluation;
    Port port;

    evaluation.next_field = current;

    evaluation.identity = FALSE;
    evaluation.result = RESULT_REJECT;
    evaluation.action = ACTION_REJECT;

    evaluation.meaning.before = current.state;
    evaluation.meaning.after = current.state;
    evaluation.meaning.identity = FALSE;
    evaluation.meaning.result = RESULT_REJECT;

    evaluation.meaning.range_min = current.range_min;
    evaluation.meaning.range_max = current.range_max;

    evaluation.meaning.boundary_min = current.boundary_min;
    evaluation.meaning.boundary_max = current.boundary_max;

    evaluation.global.target = global_target;
    evaluation.global.loss = 0.0;
    evaluation.global.threshold = global_threshold;

    /*
       tau <= 0 は更新不能として REJECT
    */
    if (tau <= 0.0) {
        return evaluation;
    }

    port = convert_port(
        delta_z,
        abln2,
        tau
    );

    /* Layer 1 : State */
    evaluation.next_field = update_field(
        current,
        port,
        dt
    );

    /* Layer 2 : Identity */
    evaluation.result = judge_identity(
        evaluation.next_field,
        &evaluation.identity
    );

    /* Layer 3 : Meaning */
    evaluation.meaning = evaluate_meaning(
        current,
        evaluation.next_field,
        evaluation.identity,
        evaluation.result
    );

    /* Global */
    evaluation.global = evaluate_global(
        evaluation.meaning,
        global_target,
        global_threshold
    );

    /* Action */
    evaluation.action = decide_action(
        evaluation.result,
        evaluation.global
    );

    return evaluation;
}


/* =========================================================
   Minimal test entry
   I/Oなし。
   終了コードだけ返す。
========================================================= */

int main(void)
{
    Field field;
    Evaluation evaluation;

    field.resolution = 1.0;

    field.state = 0.0;

    field.range_min = -0.5;
    field.range_max = 0.5;

    field.boundary_min = -0.5;
    field.boundary_max = 0.5;

    field.step = 0.1;

    evaluation = import_port(
        field,

        0.20,                 /* delta_z */
        0.6931471805599453,   /* abln2 = ln(2) */
        1.0,                  /* tau */
        0.1,                  /* dt */

        0.0,                  /* Global.Target */
        0.10                  /* Global.Threshold */
    );

    /*
       0 = PRESERVE
       1 = REFRAME
       2 = REJECT
    */
    return (int)evaluation.action;
}
```

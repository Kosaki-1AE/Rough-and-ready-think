```Python
import numpy as np

X = np.array([
    [0, 0],
    [0, 1],
    [1, 0],
    [1, 1]
])
T = np.array([0, 0, 0, 1])
w = np.zeros(2)
b = 0.0
lr = 0.1

for epoch in range(20):
    for x, t in zip(X, T):

        z = np.dot(w, x) + b
        y = 1 if z >= 0 else 0

        error = t - y

        w += lr * error * x
        b += lr * error

print("weight:", w)
print("bias:", b)

for x in X:
    z = np.dot(w, x) + b
    y = 1 if z >= 0 else 0

    print(x, "->", y)
```

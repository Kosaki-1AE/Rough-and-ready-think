    value = (

def absorb_reference():

        "to": target,
        "entered_at": time.monotonic(),
        "returnable": True,
        "locked_reason": None,
        "success": False
    })

    current_backroom = target

    print(
        "RELEASE ->",
        "R" + room_octal(source),
        "->",
        "R" + room_octal(target)
    )

    print("MEMORY SAVED ->", history[-1]["memory_hex"])

    reset_real(hierarchy=0)

def rollback():
    global current_backroom

    if not history:
        print("Return Failed -> no history")
        return

    last = history[-1]

    if current_backroom != last["to"]:
        print("Return Failed -> route mismatch")
        return

    elapsed = time.monotonic() - last["entered_at"]

    if elapsed > RETURN_WINDOW:
        last["returnable"] = False
        last["locked_reason"] = "timeout"

    if not last["returnable"]:
        print(
            "Return Failed -> Permanently Locked:",
            last["locked_reason"]
        )
        return

    history.pop()
    current_backroom = last["from"]

    reset_real(hierarchy=0)

    print("Rollback ->", "R" + room_octal(current_backroom))
    print("elapsed ->", round(elapsed, 2), "sec")

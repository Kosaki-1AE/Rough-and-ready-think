```python
import numpy as np

class Perceptron: #言わば設計図ね、このclassっての。

    def __init__(self, n_features, learning_rate=0.1): #初期状態
        self.lr = learning_rate # 学習率
        self.w_main = 0.0 # 本命用
        self.b_main = 0.0
        self.w_backup = 0.0 # 保険用
        self.b_backup = 0.0
        
    def activation(self, z):
        return 1 if z >= 0 else 0 # 活性化関数（ステップ関数）
        
    def predict(self, x): # 責任が切り替わる境界
        z = np.dot(self.w, x) + self.b
        return self.activation(z) # 予測

    def choose(self, main, backup):
	    main_score = np.dot(self.w_main, main) + self.b_main
	    if main_score >= 0:
	        return main, "main"
	    else:
	        return backup, "backup"

    def train_one(self, x, y_main, y_backup): # 1データ分の学習
        x_main = x[0]
        x_backup = x[1]
        
        main_score = self.w_main * x_main + self.b_main
        main_pred = self.activation(main_score)
        
        error_main = y_main - main_pred
	    
        if main_pred == y_main:
	        error_backup = 0
	        selected = "main"
	        
        else:
            backup_score = ( np.dot(self.w_backup, x_backup) + self.b_backup)
            backup_pred = self.activation(backup_score)
            error_backup = y_backup - backup_pred
            selected = "backup"
        
        self.w_main += self.lr * error_main * x_main
        self.b_main += self.lr * error_main
        self.w_backup += self.lr * error_backup * x_backup
        self.b_backup += self.lr * error_backup

        return error_main, error_backup, selected

    def fit(self, X, y_main, y_backup, epochs=10): # データセット全体を学習
        for epoch in range(epochs):
            errors = 0
            
            for xi, ym, yb in zip(X, y_main, y_backup):
                error_main, error_backup, selected = self.train_one(xi, ym, yb)
                
                if error_main != 0 or error_backup != 0:
                    errors += 1
            print(
				f"epoch={epoch + 1}, "
				f"errors={errors}, "
				f"w_main={self.w_main:.2f}, "
                f"b_main={self.b_main:.2f}, "
                f"w_backup={self.w_backup:.2f}, "
                f"b_backup={self.b_backup:.2f}"
			)
# print("Perceptron loaded!!")
X = np.array([ #与えられる関数
    [0, 0],
    [0, 1],
    [1, 0],
    [1, 1]
])

#答えとする解答
y_main = np.array([0, 1, 1, 1])
y_backup = np.array([0, 1, 0, 1])

model = Perceptron(
	n_features=2,
	learning_rate=0.1
) #実体化

model.fit(X, y_main, y_backup, epochs=10)
```

perseptronの方：
- Perceptronを作った瞬間に最初の状態を設定する
- 初期状態で「全部1側にいる」と仮定して学習開始する
- 誤差をどの入力方向へ返すか
- weight = 各要素への責任割当 / bias = 系全体の判定基準の補正
- Perceptronを作る -> fitを呼ぶ -> train_oneを呼ぶ  -> predictを呼ぶ -> activationを呼ぶ -> 戻ってweight/bias更新

俺の方：
- 責任を何として保持するか
- 基準線をどう定義するか
- 何を誤差と呼ぶか
- 誤差をどう責任へ配分するか
- 本命 = 第一候補 / 保険 = 第二候補
- bias = 本命を採用する最低ライン
- weight = 各候補への責任配分
- error = 選択結果に対する修正信号

モテるモテない論争とかもここにぶち込めるかもしらん。

```Python
import numpy as np

class Perceptron:

    def __init__(self, n_features, learning_rate=0.1):
        self.w = np.zeros(n_features)   # 重み
        self.b = 0.0                    # バイアス
        self.lr = learning_rate         # 学習率


    def activation(self, z):
        return 1 if z >= 0 else 0 # ステップ関数


    def predict(self, x):
        z = np.dot(self.w, x) + self.b # 重み付き和 + バイアス
        return self.activation(z) # 0 or 1 に変換


    def train_one(self, x, y):
        y_pred = self.predict(x) # 予測
        error = y - y_pred # 誤差
        self.w += self.lr * error * x # 重み更新
        self.b += self.lr * error # バイアス更新
        return error


    def fit(self, X, y, epochs=10):
    
        for epoch in range(epochs):
            errors = 0
            
            for xi, yi in zip(X, y):
                error = self.train_one(xi, yi)

                if error != 0:
                    errors += 1

            print(
                f"epoch={epoch + 1}, "
                f"errors={errors}, "
                f"w={self.w}, "
                f"b={self.b}"
            )

X = np.array([
    [0, 0],
    [0, 1],
    [1, 0],
    [1, 1]
])

y = np.array([
    0,
    0,
    0,
    1
])

model = Perceptron(
    n_features=2,
    learning_rate=0.1
)


model.fit(
    X,
```

- Perceptronを作る -> fitを呼ぶ -> train_oneを呼ぶ  -> predictを呼ぶ -> activationを呼ぶ -> 戻ってweight/bias更新

- error = 選択結果に対する修正信号

これかぁ：
- x_main = 本命候補の入力状態(0/1)
- x_backup = 保険候補の入力状態(0/1)
- x_m_main = 本命候補の中での本命候補の入力状態
- x_b_main = 保険候補の中での本命候補の入力状態
- x_m_backup = x_b_backup = 候補集合
- y_main = 本命候補が条件を満たしたか
- y_backup = 保険候補が条件を満たしたか
- w_main / w_backup = 各候補に対する責任重み
- b_main / b_backup = 各候補を採用するための基準線
- error_main / error_backup = 各候補の予測と結果のズレ

んでこうかぁ：
- x = x_main $\lor$ x_backup
- x_m_main $\lor$ x_b_main = x_main
- x_m_backup $\lor$ x_b_backup = x_backup
- $\neg$ x_main -> x_backup
- $\neg$ y_main -> y_backup
- x_backup -(x_m_backup・x_b_backup)> y_backup
- x_m_backup・x_b_backup -(w_main / w_backup)> r_sum
- r_sum -(w_main / w_backup)> b_main / b_backup
- error_main / error_backup -(x_m_main・x_b_main)> x
- x = x_main + x_backup：reset

```python
import numpy as np

class Perceptron:

    def __init__(self, n_features=2, learning_rate=0.1):
        self.lr = learning_rate
        self.w = np.zeros(n_features)
        self.b = 0.0
        self.w_main = [0.0]
        self.b_main = [0.0]
        self.w_backup = [np.zeros(2)]
        self.b_backup = [0.0]
        self.memory = []
        self.backup_states = [
            np.array([0, 0]),
            np.array([0, 1]),
            np.array([1, 0]),
            np.array([1, 1])
        ]
        self.backup_index = 0
    
    def next_backup(self):
        backup = self.backup_states[
            self.backup_index
        ]
        self.backup_index += 1
        if self.backup_index >= len(self.backup_states):
            self.backup_index = 0
        return backup
    
    def activation(self, z):
        return 1 if z >= 0 else 0

    def logical_or(self, a, b):
        return int(bool(a) or bool(b))

    def train_one(self, x, y_main, y_backup, epoch, sample_id):
        x_main = int(x[0])
        x_backup = int(x[1])
        x_total = self.logical_or(x_main, x_backup)
        main_score = (self.w_main[-1] * x_main + self.b_main[-1])
        main_pred = self.activation(main_score)
        error_main = (y_main - main_pred)

        if main_pred == y_main:
            selected = "main"
            backup = None
            backup_score = None
            backup_pred = None
            error_backup = 0

        else:
            backup = self.next_backup()
            selected = "backup"
            backup_main = backup[0]
            backup_sub = backup[1]
            backup_score = np.dot(self.w_backup[-1], backup) + self.b_backup[-1]
            backup_pred = self.activation(backup_score)
            error_backup = y_backup - backup_pred

        old_w_main = self.w_main[-1]
        old_b_main = self.b_main[-1]
        old_w_backup = self.w_backup[-1].copy()
        old_b_backup = self.b_backup[-1]
        new_b_main = (self.b_main[-1] + self.lr * error_main)
        new_w_main = (self.w_main[-1] + new_b_main * x_main)
        new_b_backup = (self.b_backup[-1] + self.lr * error_backup)
        if backup is not None:
            new_w_backup = (self.w_backup[-1] + new_b_backup * backup)
        else:
            new_w_backup = (self.w_backup[-1].copy())
        self.b_main.append(new_b_main)
        self.w_main.append(new_w_main)
        self.b_backup.append(new_b_backup)
        self.w_backup.append(new_w_backup)

        self.memory.append({
            "epoch": epoch,
            "sample_id": sample_id, # 時間
            
            "x_main": x_main,
            "x_backup": x_backup, # OR前
            "x": x_total, # OR後
            
            "y_main": y_main,
            "y_backup": y_backup, # 教師
            
            "main_score": main_score,
            "main_pred": main_pred, # 予測
            "error_main": error_main,
            
            "backup_state":
                None
                if backup is None
                else backup.copy(),
            "backup_score": backup_score,
            "backup_pred": backup_pred,
            "error_backup": error_backup, # 誤差
            
            "selected": selected, # 選択経路
            
            "old_w_main": old_w_main, # 更新前
            "old_b_main": old_b_main,
            "old_w_backup": old_w_backup,
            "old_b_backup": old_b_backup,

            "w_main": self.w_main[-1], # 更新後
            "b_main": self.b_main[-1],
            "w_backup": self.w_backup[-1].copy(),
            "b_backup": self.b_backup[-1]
        })

        return (error_main, error_backup, selected)

    def fit(self, X, y_main, y_backup, epochs=10):
        for epoch in range(epochs):
            errors = 0

            for sample_id, (xi, ym, yb) in enumerate(zip(X, y_main, y_backup)):
                (error_main, error_backup, selected) = self.train_one(xi, ym, yb, epoch + 1, sample_id)

                if (error_main != 0 or error_backup != 0):
                    errors += 1

            print(
                f"epoch={epoch + 1}, "
                f"errors={errors}, "
                f"w_main={self.w_main[-1]:.2f}, "
                f"b_main={self.b_main[-1]:.2f}, "
                f"w_backup={self.w_backup[-1]}, "
                f"b_backup={self.b_backup[-1]:.2f}"
            )

    def show_memory(self):
        print("\n===== MEMORY =====")
        for m in self.memory:
            print(
                f"epoch={m['epoch']} "
                f"sample={m['sample_id']} | "
                f"x_main={m['x_main']} "
                f"x_backup={m['x_backup']} "
                f"-> x={m['x']} | "
                f"selected={m['selected']} | "
                f"backup_state={m['backup_state']} | "
                f"error_main={m['error_main']} "
                f"error_backup={m['error_backup']}"
            )

X = np.array([
    [0, 0],
    [0, 1],
    [1, 0],
    [1, 1]
])

y_main = np.array([0, 1, 1, 1])
y_backup = np.array([1, 1, 1, 1])

model = Perceptron(

    learning_rate=0.1
)

model.fit(
```

```python
import numpy as np
import sympy as sp

class DynamicalSystem:

    def __init__(self):
        self.memory = []
        print("=== Dynamical System ===")
        self.x = float(input("初期値 x0 を入力してください: ")) # 初期値
        self.r = float(input("パラメータ r を入力してください: ") # パラメータ        )
        equation_text = input(
            "x(t+1) の式を入力してください\n"
            "例: r*x*(1-x)\n"
            "> "
        ) # 式

        x, r = sp.symbols("x r") # 記号を作る

        expression = sp.sympify(equation_text) # 文字列 → 数式

        # 数式 → Pythonで計算できる関数
        self.equation = sp.lambdify((x, r), expression, "numpy")
        # 初期状態をMemoryへ
        self.memory.append(self.x)


    def step(self):
        next_x = self.equation(self.x, self.r)
        self.x = next_x
        self.memory.append(next_x)
        return next_x

    def run(self, steps=100):
        for _ in range(steps):
            self.step()
        return self.memory

model = DynamicalSystem()
result = model.run(
    steps=100
)

for t, x in enumerate(result):
    print(
        f"t={t}, "
        f"x={x}"
    )
```

```python
import numpy as np
# =========================================
# 1. 状態履歴
# =========================================
class StateHistory:

    def __init__(self, initial_value):
        self.values = [initial_value]

    def current(self):
        return self.values[-1]

    def add(self, value):
        self.values.append(value)

    def all(self):
        return self.values
# =========================================
# 2. Memory
# =========================================
class Memory:

    def __init__(self):
        self.data = []

    def store(self, state):
        self.data.append(state)

    def all(self):
        return self.data
# =========================================
# 3. Backup候補探索
# =========================================
class BackupSelector:

    def __init__(self):
        self.states = [
            np.array([0, 0]),
            np.array([0, 1]),
            np.array([1, 0]),
            np.array([1, 1])
        ]
        self.index = 0

    def next(self):
        state = self.states[self.index]
        self.index += 1
        if self.index >= len(self.states):
            self.index = 0
        return state.copy()
# =========================================
# 4. Perceptron本体
# =========================================
class Perceptron:

    def __init__(
        self,
        n_features=2,
        learning_rate=0.1
    ):
        self.lr = learning_rate
        # 元のPerceptron
        self.w = np.zeros(n_features)
        self.b = 0.0
        # main
        self.w_main = StateHistory(0.0)
        self.b_main = StateHistory(0.0)
        # backup
        self.w_backup = StateHistory(
            np.zeros(2)
        )
        self.b_backup = StateHistory(0.0)
        # Memory
        self.memory = Memory()
        # Backup探索器
        self.backup = BackupSelector()


    def activation(self, z):
        return 1 if z >= 0 else 0


    def logical_or(self, a, b):
        return int(bool(a) or bool(b))

    def train_one(self, x, y_main, y_backup, epoch, sample_id):
        x_main = int(x[0])
        x_backup = int(x[1])
        x_total = self.logical_or(x_main, x_backup)

        # =================================
        # main
        # =================================

        main_score = (self.w_main.current() * x_main + self.b_main.current())
        main_pred = self.activation(main_score)
        error_main = (y_main - main_pred)

        # =================================
        # 選択
        # =================================

        if main_pred == y_main:
            selected = "main"
            backup_state = None
            backup_score = None
            backup_pred = None
            error_backup = 0

        else:
            selected = "backup"
            backup_state = (self.backup.next())
            backup_score = (
                np.dot(self.w_backup.current(), backup_state)
                + self.b_backup.current()
            )
            backup_pred = self.activation(backup_score)
            error_backup = (y_backup - backup_pred)

        # =================================
        # main更新
        # =================================

        new_b_main = (self.b_main.current() + self.lr * error_main)
        new_w_main = (self.w_main.current() + new_b_main * x_main)
        self.b_main.add(new_b_main)
        self.w_main.add(new_w_main)

        # =================================
        # backup更新
        # =================================

        new_b_backup = (self.b_backup.current() + self.lr * error_backup)

        if backup_state is not None:
            new_w_backup = (
                self.w_backup.current() + new_b_backup * backup_state
            )

        else:
            new_w_backup = (
                self.w_backup.current().copy()
            )

        self.b_backup.add(new_b_backup)
        self.w_backup.add(new_w_backup)

        # =================================
        # Memory
        # =================================

        self.memory.store({
            "epoch": epoch,
            
            "sample_id": sample_id,
            
            "x_main": x_main,
            "x_backup": x_backup,

            "x": x_total,

            "y_main": y_main,
            "y_backup": y_backup,

            "main_pred": main_pred,

            "backup_state": backup_state,
            "backup_pred": backup_pred,
            "error_main": error_main,
            "error_backup": error_backup,
            "selected": selected,

            "w_main": self.w_main.current(),
            "b_main": self.b_main.current(),
            "w_backup": self.w_backup.current().copy(),
            "b_backup": self.b_backup.current()
        })

        return (error_main, error_backup, selected)

    def fit(self, X, y_main, y_backup, epochs=10):
        for epoch in range(epochs):
            errors = 0
            
            for sample_id, (xi, ym, yb) in enumerate(zip(X, y_main, y_backup)):
                (error_main, error_backup, selected) = self.train_one(
                    xi, ym, yb, epoch + 1, sample_id
                )
                if (error_main != 0 or error_backup != 0):
                    errors += 1

            print(
                f"epoch={epoch + 1}, "
                f"errors={errors}, "
                f"w_main={self.w_main.current():.2f}, "
                f"b_main={self.b_main.current():.2f}, "
                f"w_backup={self.w_backup.current()}, "
                f"b_backup={self.b_backup.current():.2f}"
            )
```
